# =====================================================
# Sampling helpers
# =====================================================


def apply_repetition_penalty(
    logits,
    tokens,
    penalty
):

    unique_tokens = torch.unique(
        tokens
    )

    logits[:, unique_tokens] /= penalty

    return logits



def top_k_top_p_filter(
    logits,
    top_k=50,
    top_p=0.95
):

    # ----------------------
    # top-k
    # ----------------------

    if top_k:

        values, indices = torch.topk(
            logits,
            top_k
        )


        filtered = torch.full_like(
            logits,
            float("-inf")
        )


        filtered.scatter_(
            1,
            indices,
            values
        )


        logits = filtered



    # ----------------------
    # nucleus sampling
    # ----------------------

    if top_p < 1.0:

        sorted_logits, sorted_indices = torch.sort(
            logits,
            descending=True
        )


        probs = torch.softmax(
            sorted_logits,
            dim=-1
        )


        cumulative_probs = torch.cumsum(
            probs,
            dim=-1
        )


        remove = cumulative_probs > top_p


        remove[:,1:] = remove[:,:-1].clone()

        remove[:,0] = False


        sorted_logits[remove] = float("-inf")


        filtered = torch.full_like(
            logits,
            float("-inf")
        )


        filtered.scatter_(
            1,
            sorted_indices,
            sorted_logits
        )


        logits = filtered


    return logits



# =====================================================
# Generation
# =====================================================


@torch.no_grad()
def generate(
    instruction,
    max_new_tokens=MAX_NEW_TOKENS
):


    prompt = (
        "### Instruction:\n"
        + instruction.strip()
        + "\n\n### Response:\n"
    )


    input_ids = tokenizer.encode(
        prompt,
        return_tensors="pt"
    ).to(
        DEVICE
    )


    generated = input_ids.clone()



    for _ in range(max_new_tokens):


        context = generated[
            :, -BLOCK_SIZE:
        ]



        with torch.autocast(
            device_type="cuda",
            dtype=torch.float16
        ):


            logits, _ = model(
                context
            )



        logits = logits[:, -1, :]



        logits /= TEMPERATURE



        logits = apply_repetition_penalty(
            logits,
            generated,
            REPETITION_PENALTY
        )



        logits = top_k_top_p_filter(
            logits,
            TOP_K,
            TOP_P
        )



        probs = torch.softmax(
            logits,
            dim=-1
        )



        next_token = torch.multinomial(
            probs,
            1
        )



        generated = torch.cat(
            [
                generated,
                next_token
            ],
            dim=1
        )



        if next_token.item() == tokenizer.eos_token_id:

            break



    decoded = tokenizer.decode(
        generated[0],
        skip_special_tokens=True
    )


    # remove instruction part

    response = decoded.split(
        "### Response:\n"
    )[-1]


    # stop if it starts another instruction

    for stop in [
        "### Instruction:",
        "User:"
    ]:

        if stop in response:

            response = response.split(
                stop
            )[0]


    return response.strip()

repeated tokens (ItItItIt, """""""")

Your validation loss of 0.5446 was misleading because the validation objective only measured token prediction on the masked labels, not whether free-form generation works.

Your training:

LR = 3e-5

Alpaca has ~52k examples.

That is a lot.

For instruction tuning a pretrained GPT, usually:

1 epoch
LR 5e-6 to 1e-5

is enough.

Your model probably overfit the instruction format and lost general language ability.
